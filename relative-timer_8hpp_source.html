104 double run()
105 {
106 // Handle special values for safety
107 if (batch <= 0) batch = 1;
108 if (loop <= 0) loop = 1;
109
110 int inner_loop = loop / batch;
111 if (inner_loop <= 0) inner_loop = 1;
112
113 // Verification phase - check functional correctness before performance testing
114 if (!methodVerify())
115 {
116 // Return NaN to indicate verification failure
117 timeA = 0.0;
118 timeB = 0.0;
119 ratio = std::numeric_limits<double>::quiet_NaN();
120 return ratio;
121 }
122
123 // Warm up
124 methodA();
125 methodB();
126
127 // Time accumulation
128 timeA = 0.0;
129 timeB = 0.0;
130
131 // Run in alternating batches
132 for (int i = 0; i < batch; ++i)
133 {
134 // Time methodA
135 auto startA = std::chrono::high_resolution_clock::now();
136 for (int j = 0; j < inner_loop; ++j)
137 {
138 methodA();
139 }
140 auto endA = std::chrono::high_resolution_clock::now();
141 std::chrono::duration<double> elapsedA = endA - startA;
142 timeA += elapsedA.count();
143
144 // Time methodB
145 auto startB = std::chrono::high_resolution_clock::now();
146 for (int j = 0; j < inner_loop; ++j)
147 {
148 methodB();
149 }
150 auto endB = std::chrono::high_resolution_clock::now();
151 std::chrono::duration<double> elapsedB = endB - startB;
152 timeB += elapsedB.count();
153 }
154
155 // Convert to milliseconds
156 timeA *= 1000.0;
157 timeB *= 1000.0;
158
159 // Calculate ratio
160 if (timeB == 0.0)
161 {
162 // Avoid division by zero
163 ratio = (timeA == 0.0) ? 1.0 : 1000.0; // Large value indicating A is much slower
164 }
165 else
166 {
167 ratio = timeA / timeB;
168 }
169
170 return ratio;
171 }
